cmake_minimum_required(VERSION 3.20)
project(Vivarium VERSION 0.2.0 LANGUAGES CXX)
option(VIVARIUM_BUILD_TESTS "Build tests" ON)
option(VIVARIUM_BUILD_EXAMPLES "Build examples" ON)
option(VIVARIUM_BUILD_PYTHON "Build optional pybind11 module" OFF)
option(VIVARIUM_BUILD_MOSS_EDITOR "Build the Moss-powered editor application" OFF)
set(VIVARIUM_MOSS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../Moss" CACHE PATH "Path to the TxbiG/Moss checkout")
add_library(vivarium src/vivarium.cpp src/recorder.cpp src/scene.cpp)
add_library(Vivarium::vivarium ALIAS vivarium)
target_compile_features(vivarium PUBLIC cxx_std_17)
target_include_directories(vivarium PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set_target_properties(vivarium PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
 target_compile_options(vivarium PRIVATE /W4)
else()
 target_compile_options(vivarium PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(VIVARIUM_BUILD_EXAMPLES)
 add_executable(vivarium_random_agent examples/cpp/random_agent.cpp)
 target_link_libraries(vivarium_random_agent PRIVATE Vivarium::vivarium)
endif()
if(VIVARIUM_BUILD_TESTS)
 enable_testing()
 add_executable(vivarium_tests tests/test_main.cpp)
 target_link_libraries(vivarium_tests PRIVATE Vivarium::vivarium)
 add_test(NAME vivarium_tests COMMAND vivarium_tests)
endif()
if(VIVARIUM_BUILD_PYTHON)
 find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
 find_package(pybind11 CONFIG REQUIRED)
 pybind11_add_module(_vivarium bindings/python/module.cpp)
 target_link_libraries(_vivarium PRIVATE Vivarium::vivarium)
 target_compile_features(_vivarium PRIVATE cxx_std_17)
endif()

if(VIVARIUM_BUILD_MOSS_EDITOR)
 if(NOT EXISTS "${VIVARIUM_MOSS_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Moss checkout not found at '${VIVARIUM_MOSS_ROOT}'. Set -DVIVARIUM_MOSS_ROOT=<path-to-TxbiG/Moss>.")
 endif()
 # Avoid the dependency's install rules changing the parent's install layout.
 set(ENABLE_INSTALL OFF CACHE BOOL "Disable nested Moss install rules" FORCE)
 set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "Disable Moss LTO for parent integration" FORCE)
 add_subdirectory("${VIVARIUM_MOSS_ROOT}" "${CMAKE_BINARY_DIR}/moss-build" EXCLUDE_FROM_ALL)
 if(NOT TARGET Moss)
  message(FATAL_ERROR "The supplied Moss checkout did not define the expected 'Moss' CMake target.")
 endif()
 add_executable(vivarium_editor
  editor/main.cpp
  editor/EditorApplication.cpp
  editor/panels/SimulationControls.cpp
  editor/panels/EnvironmentInspector.cpp
  editor/panels/SceneBuilderPanel.cpp
  editor/panels/MetricsPanel.cpp)
 target_compile_features(vivarium_editor PRIVATE cxx_std_17)
 target_include_directories(vivarium_editor PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/editor" "${VIVARIUM_MOSS_ROOT}/include")
 target_link_libraries(vivarium_editor PRIVATE Vivarium::vivarium Moss)
endif()
include(GNUInstallDirs)
install(TARGETS vivarium EXPORT VivariumTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
